#!/bin/sh
# MedSyn Web Server
# IP: 10.10.10.13/26, GW: 10.10.10.1

hostname SMLC-WEB1

ip addr flush dev eth0
ip addr add 10.10.10.13/26 dev eth0
ip link set eth0 up
ip route add default via 10.10.10.1
echo "nameserver 10.10.10.12" > /etc/resolv.conf

apk update && apk add lighttpd

mkdir -p /var/www/localhost/htdocs
cat > /var/www/localhost/htdocs/index.html << 'HTMLEOF'
<!DOCTYPE html>
<html>
<head><title>MedSyn Company - Internal Web Portal</title></head>
<body>
<h1>MedSyn Company Internal Web Portal</h1>
<p>Server: WEB-SRV | IP: 10.10.10.13 | Zone: DMZ</p>
</body>
</html>
HTMLEOF

rc-update add lighttpd default
rc-service lighttpd start
